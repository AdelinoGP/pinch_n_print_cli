//! [T25-PROBE] Temporary probe test (wayfinder ticket 25).
//!
//! Question: `point_in_contour_winding(ring, x, y, eps)` runs two passes —
//! (1) a per-edge point-to-segment projection under `dist2 <= eps²`, then
//! (2) the winding scan. The Arachne `PerimeterSpatialContext` queries all
//! pass `eps_mm = 0.0`, for which pass (1) can only fire on an exactly-on-edge
//! point but still costs a full projection per edge. How much of the query is
//! pass (1)?
//!
//! Method: time the real predicate against a winding-only replica on the same
//! rings and query points.
//!
//! Probe-only: delete before any commit.
//!
//! Run: cargo test -p slicer-core --features host-algos --test t25_probe_winding_pass_share_tdd

#![cfg(feature = "host-algos")]

use slicer_ir::{point_in_contour_winding, Polygon};

fn ring(cx: i64, cy: i64, r: i64, n: usize) -> Polygon {
    let points = (0..n)
        .map(|i| {
            let theta = (i as f64) * std::f64::consts::TAU / (n as f64);
            slices_pt(
                cx + (r as f64 * theta.cos()) as i64,
                cy + (r as f64 * theta.sin()) as i64,
            )
        })
        .collect();
    Polygon { points }
}

fn slices_pt(x: i64, y: i64) -> slicer_ir::Point2 {
    slicer_ir::Point2 { x, y }
}

/// Winding-only replica: the second pass of `point_in_contour_winding`.
fn winding_only(ring: &Polygon, px_mm: f64, py_mm: f64) -> bool {
    let pts = &ring.points;
    let n = pts.len();
    if n < 3 {
        return false;
    }
    let mut wn: i32 = 0;
    for i in 0..n {
        let a = pts[i];
        let b = pts[(i + 1) % n];
        let ax = a.x as f64 / 10_000.0;
        let ay = a.y as f64 / 10_000.0;
        let bx = b.x as f64 / 10_000.0;
        let by = b.y as f64 / 10_000.0;
        if ay <= py_mm {
            if by > py_mm && (bx - ax) * (py_mm - ay) - (px_mm - ax) * (by - ay) > 0.0 {
                wn += 1;
            }
        } else if by <= py_mm && (bx - ax) * (py_mm - ay) - (px_mm - ax) * (by - ay) < 0.0 {
            wn -= 1;
        }
    }
    wn != 0
}

#[test]
fn boundary_pre_pass_share_of_the_winding_predicate() {
    // A band polygon as the overhang producer emits: a many-vertex closed ring
    // (1 mm radius circle ring is representative of one band polygon).
    for n in [64usize, 512, 4096] {
        let ring = ring(0, 0, 10_000, n); // 1 mm radius in units
        let pts: Vec<(f64, f64)> = (0..64)
            .map(|i| {
                let t = i as f64 / 64.0;
                (t * 0.9, -t * 0.9)
            })
            .collect();

        let reps = 2_000;
        let start = std::time::Instant::now();
        let mut hits_full = 0usize;
        for _ in 0..reps {
            for &(x, y) in &pts {
                if point_in_contour_winding(&ring, x, y, 0.0) {
                    hits_full += 1;
                }
            }
        }
        let full_ns = start.elapsed().as_nanos() as f64 / (reps * pts.len()) as f64;

        let start = std::time::Instant::now();
        let mut hits_wind = 0usize;
        for _ in 0..reps {
            for &(x, y) in &pts {
                if winding_only(&ring, x, y) {
                    hits_wind += 1;
                }
            }
        }
        let wind_ns = start.elapsed().as_nanos() as f64 / (reps * pts.len()) as f64;

        println!(
            "PROBE n={n} full_ns_per_call={full_ns:.1} winding_only_ns={wind_ns:.1} \
             pass1_share={:.3} hits_full={hits_full} hits_wind={hits_wind}",
            (full_ns - wind_ns) / full_ns
        );
    }
}
